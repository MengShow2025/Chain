# TitanChain 多语言国际化实现指南

## 1. 项目依赖安装

### 1.1 核心依赖
```bash
# 安装react-i18next和相关依赖
pnpm add react-i18next i18next i18next-browser-languagedetector i18next-http-backend

# 安装类型定义
pnpm add -D @types/react-i18next
```

### 1.2 依赖说明
- `react-i18next`: React的i18next绑定库
- `i18next`: 核心国际化框架
- `i18next-browser-languagedetector`: 浏览器语言检测插件
- `i18next-http-backend`: HTTP后端加载插件

```typescript
import { initReactI18next } from 'react-i18next';
import LanguageDetector from 'i18next-browser-languagedetector';
import HttpBackend from 'i18next-http-backend';

// 导入所有语言资源

// 其他语言资源导入...

export const SUPPORTED_LANGUAGES = [
  {

    nativeName: 'English',
    flag: '🇺🇸',
    rtl: false
  },
  {
    code: 'zh',
    name: 'Chinese',
    nativeName: '中文',
    flag: '🇨🇳',
    rtl: false
  },
  {
    code: 'ja',
    name: 'Japanese',
    nativeName: '日本語',
    flag: '🇯🇵',
    rtl: false
  },
  {
    code: 'ko',
    name: 'Korean',
    nativeName: '한국어',
    flag: '🇰🇷',
    rtl: false
  },
  {
    code: 'pt',
    name: 'Portuguese',
    nativeName: 'Português',
    flag: '🇵🇹',
    rtl: false
  },
  {
    code: 'ar',
    name: 'Arabic',
    nativeName: 'العربية',
    flag: '🇸🇦',
    rtl: true
  }
];

// 语言资源配置
const resources = {
  en: {
    common: enCommon,
    home: enHome,
    explorer: enExplorer,
    wallet: enWallet,
    staking: enStaking,
    governance: enGovernance,
    validators: enValidators
  },
  zh: {
    common: zhCommon,
    home: zhHome,
    explorer: zhExplorer,
    wallet: zhWallet,
    staking: zhStaking,
    governance: zhGovernance,
    validators: zhValidators
  }
  // 其他语言资源...
};

// i18n初始化配置
i18n
  .use(HttpBackend)
  .use(LanguageDetector)
  .use(initReactI18next)
  .init({
    resources,
    fallbackLng: 'en',
    debug: process.env.NODE_ENV === 'development',
    
    // 语言检测配置
    detection: {
      order: ['localStorage', 'navigator', 'htmlTag'],
      caches: ['localStorage'],
      lookupLocalStorage: 'titanchain-language'
    },
    
    // 插值配置
    interpolation: {
      escapeValue: false,
      format: (value, format, lng) => {
        if (format === 'currency') {
          return new Intl.NumberFormat(lng, {
            style: 'currency',
            currency: 'USD'
          }).format(value);
        }
        if (format === 'number') {
          return new Intl.NumberFormat(lng).format(value);
        }
        if (format === 'percent') {
          return new Intl.NumberFormat(lng, {
            style: 'percent'
          }).format(value);
        }
        if (format === 'datetime') {
          return new Intl.DateTimeFormat(lng, {
            year: 'numeric',
            month: 'short',
            day: 'numeric',
            hour: '2-digit',
            minute: '2-digit'
          }).format(new Date(value));
        }
        return value;
      }
    },
    
    // 命名空间配置
    ns: ['common', 'home', 'explorer', 'wallet', 'staking', 'governance', 'validators'],
    defaultNS: 'common'
  });

export default i18n;
```

## 4. 组件实现

### 4.1 语言选择器组件 (src/components/LanguageSelector.tsx)
```typescript
import React, { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { ChevronDown, Globe } from 'lucide-react';
import { SUPPORTED_LANGUAGES } from '../i18n';
import type { LanguageOption } from '../i18n/types';

interface LanguageSelectorProps {
  className?: string;
  variant?: 'dropdown' | 'modal';
}

export const LanguageSelector: React.FC<LanguageSelectorProps> = ({
  className = '',
  variant = 'dropdown'
}) => {
  const { i18n } = useTranslation();
  const [isOpen, setIsOpen] = useState(false);
  
  const currentLanguage = SUPPORTED_LANGUAGES.find(
    lang => lang.code === i18n.language
  ) || SUPPORTED_LANGUAGES[0];
  
  const handleLanguageChange = async (language: LanguageOption) => {
    await i18n.changeLanguage(language.code);
    setIsOpen(false);
    
    // 更新HTML dir属性以支持RTL
    document.documentElement.dir = language.rtl ? 'rtl' : 'ltr';
    document.documentElement.lang = language.code;
  };
  
  if (variant === 'modal') {
    return (
      <div className={`relative ${className}`}>
        <button
          onClick={() => setIsOpen(!isOpen)}
          className="flex items-center gap-2 px-4 py-2 bg-white/10 backdrop-blur-sm rounded-lg border border-white/20 hover:bg-white/20 transition-all duration-200"
        >
          <Globe className="w-4 h-4" />
          <span className="text-sm font-medium">{currentLanguage.flag}</span>
          <ChevronDown className={`w-4 h-4 transition-transform ${isOpen ? 'rotate-180' : ''}`} />
        </button>
        
        {isOpen && (
          <>
            <div 
              className="fixed inset-0 z-40" 
              onClick={() => setIsOpen(false)}
            />
            <div className="absolute top-full mt-2 right-0 z-50 bg-white dark:bg-gray-800 rounded-lg shadow-xl border border-gray-200 dark:border-gray-700 min-w-[200px]">
              {SUPPORTED_LANGUAGES.map((language) => (
                <button
                  key={language.code}
                  onClick={() => handleLanguageChange(language)}
                  className={`w-full flex items-center gap-3 px-4 py-3 text-left hover:bg-gray-50 dark:hover:bg-gray-700 transition-colors first:rounded-t-lg last:rounded-b-lg ${
                    currentLanguage.code === language.code 
                      ? 'bg-blue-50 dark:bg-blue-900/20 text-blue-600 dark:text-blue-400' 
                      : 'text-gray-700 dark:text-gray-300'
                  }`}
                >
                  <span className="text-lg">{language.flag}</span>
                  <div className="flex flex-col">
                    <span className="font-medium">{language.nativeName}</span>
                    <span className="text-xs text-gray-500 dark:text-gray-400">{language.name}</span>
                  </div>
                </button>
              ))}
            </div>
          </>
        )}
      </div>
    );
  }
  
  return (
    <div className={`relative ${className}`}>
      <select
        value={currentLanguage.code}
        onChange={(e) => {
          const language = SUPPORTED_LANGUAGES.find(lang => lang.code === e.target.value);
          if (language) handleLanguageChange(language);
        }}
        className="appearance-none bg-white/10 backdrop-blur-sm border border-white/20 rounded-lg px-3 py-2 pr-8 text-sm font-medium focus:outline-none focus:ring-2 focus:ring-blue-500"
      >
        {SUPPORTED_LANGUAGES.map((language) => (
          <option key={language.code} value={language.code}>
            {language.flag} {language.nativeName}
          </option>
        ))}
      </select>
      <ChevronDown className="absolute right-2 top-1/2 transform -translate-y-1/2 w-4 h-4 pointer-events-none" />
    </div>
  );
};
```

### 4.2 RTL布局包装器 (src/components/RTLWrapper.tsx)
```typescript
import React from 'react';
import { useTranslation } from 'react-i18next';
import { SUPPORTED_LANGUAGES } from '../i18n';

interface RTLWrapperProps {
  children: React.ReactNode;
  className?: string;
}

export const RTLWrapper: React.FC<RTLWrapperProps> = ({ 
  children, 
  className = '' 
}) => {
  const { i18n } = useTranslation();
  
  const currentLanguage = SUPPORTED_LANGUAGES.find(
    lang => lang.code === i18n.language
  );
  
  const isRTL = currentLanguage?.rtl || false;
  
  return (
    <div 
      className={`${className} ${isRTL ? 'rtl' : 'ltr'}`}
      dir={isRTL ? 'rtl' : 'ltr'}
    >
      {children}
    </div>
  );
};
```

### 4.3 语言管理Hook (src/hooks/useLanguage.ts)
```typescript
import { useTranslation } from 'react-i18next';
import { SUPPORTED_LANGUAGES } from '../i18n';
import type { LanguageOption } from '../i18n/types';

export const useLanguage = () => {
  const { i18n } = useTranslation();
  
  const currentLanguage = SUPPORTED_LANGUAGES.find(
    lang => lang.code === i18n.language
  ) || SUPPORTED_LANGUAGES[0];
  
  const changeLanguage = async (languageCode: string) => {
    const language = SUPPORTED_LANGUAGES.find(lang => lang.code === languageCode);
    if (!language) return;
    
    await i18n.changeLanguage(languageCode);
    
    // 更新HTML属性
    document.documentElement.dir = language.rtl ? 'rtl' : 'ltr';
    document.documentElement.lang = languageCode;
    
    // 保存到localStorage
    localStorage.setItem('titanchain-language', languageCode);
  };
  
  const isRTL = currentLanguage.rtl;
  
  return {
    currentLanguage,
    supportedLanguages: SUPPORTED_LANGUAGES,
    changeLanguage,
    isRTL,
    isReady: i18n.isInitialized
  };
};
```

### 6.2 App组件修改 (src/App.tsx)
```typescript
import React, { Suspense } from 'react';
import { BrowserRouter as Router, Routes, Route } from 'react-router-dom';
import { useTranslation } from 'react-i18next';
import { RTLWrapper } from './components/RTLWrapper';
import { LanguageSelector } from './components/LanguageSelector';
import Home from './pages/Home';
import Explorer from './pages/Explorer';
// 其他页面导入...

function App() {
  const { t, ready } = useTranslation();
  
  if (!ready) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-lg">{t('common.loading')}</div>
      </div>
    );
  }
  
  return (
    <RTLWrapper>
      <Router>
        <div className="min-h-screen bg-gradient-to-br from-blue-900 via-purple-900 to-indigo-900">
          {/* 语言选择器 */}
          <div className="fixed top-4 right-4 z-50">
            <LanguageSelector variant="modal" />
          </div>
          
          <Suspense fallback={
            <div className="min-h-screen flex items-center justify-center">
              <div className="text-lg text-white">{t('common.loading')}</div>
            </div>
          }>
            <Routes>
              <Route path="/" element={<Home />} />
              <Route path="/explorer" element={<Explorer />} />
              {/* 其他路由... */}
            </Routes>
          </Suspense>
        </div>
      </Router>
    </RTLWrapper>
  );
}

export default App;
```
